class ContrastPlan:
    """A contrast that survived validation, with its replicate unit already decided."""

    spec: ContrastSpec
    level_a: str
    level_b: str
    model: str  # "paired" or "unpaired"
    block_unit: str  # "donor_id" or "sample_id"
    rows: tuple[str, ...]  # replicate-unit keys, in the order they enter every matrix
    blocks: tuple[tuple[str, tuple[int, ...]], ...]  # permutation blocks -> row indices
    reachable_permutations: int
    contributing: tuple[str, ...]
    block_of: dict[str, str]
    level_of: dict[str, str]
    markers: tuple[str, ...]
    populations: tuple[str, ...]

    @property
    def n_samples(self) -> int:
        return len(self.contributing)

    @property
    def n_donors(self) -> int:
        return len(self.rows)

    @property
    def comparison_id(self) -> str:
        return (
            f"contrast:{self.spec.contrast_id}:{self.spec.factor}:"
            f"{self.level_a}__vs__{self.level_b}:{self.model}"
        )

    One prespecified biological contrast.

    A contrast is written down before the data are looked at: which factor, which two levels, in which order, optionally which way the effect is expected to point, and which covariates the claim is conditional on. Whatever the design cannot support is refused by :func:resolve_contrast rather than quietly dropped.

    level_a is always the alphabetically first of the two levels. The normalisation is deliberate and matches the OpenProblems functional-marker component: with a sorted order a positive effect size means the same biological thing in every run and a sign cannot flip because two samples happened to be listed the other way round. expected_direction is stated in that normalised order.

    class ContrastSpec(BaseModel):
    """One prespecified biological contrast.

    A contrast is written down *before* the data are looked at: which factor, which two
    levels, in which order, optionally which way the effect is expected to point, and
    which covariates the claim is conditional on.  Whatever the design cannot support is
    refused by :func:`resolve_contrast` rather than quietly dropped.

    ``level_a`` is always the alphabetically first of the two levels.  The normalisation
    is deliberate and matches the OpenProblems functional-marker component: with a
    sorted order a positive effect size means the same biological thing in every run and
    a sign cannot flip because two samples happened to be listed the other way round.
    ``expected_direction`` is stated in that normalised order.
    """

    model_config = ConfigDict(frozen=True, extra="forbid")

    contrast_id: str
    factor: str
    level_a: str
    level_b: str
    expected_direction: Literal["none", "increase_in_a", "increase_in_b"] = "none"
    #: Covariates the claim is conditional on.  One nested inside a contrast level
    #: confounds the contrast with the covariate and refuses the contrast.
    protected_covariates: tuple[str, ...] = ()
    #: Restrict to these markers, usually the panel's functional markers.  ``None`` means
    #: every evaluation marker on the frozen partition.
    markers: tuple[str, ...] | None = None
    #: Restrict to these frozen populations.  ``None`` means every frozen population.
    populations: tuple[str, ...] | None = None
    #: ``auto`` matches the samples inside a donor when the donor factor allows it.
    model: Literal["auto", "unpaired", "paired"] = "auto"
    min_samples_per_level: int = MIN_SAMPLES_PER_LEVEL
    n_resamples: int = DEFAULT_N_RESAMPLES
    alpha: float = 0.05
    seed: int = 0

    @model_validator(mode="before")
    @classmethod
    def _normalise_level_order(cls, data):
        if isinstance(data, dict) and data.get("level_a") and data.get("level_b"):
            first, second = sorted((str(data["level_a"]), str(data["level_b"])))
            data = {**data, "level_a": first, "level_b": second}
        return data

    @model_validator(mode="after")
    def _check_declared(self):
        if self.level_a == self.level_b:
            raise ValueError(
                f"contrast {self.contrast_id!r}: a contrast needs two different levels, "
                f"got {self.level_a!r} twice"
            )
        if self.n_resamples < 1:
            raise ValueError("n_resamples must be at least 1")
        if not 0.0 < self.alpha < 1.0:
            raise ValueError("alpha must lie strictly between 0 and 1")
        return self

    @property
    def interval_level(self) -> int:
        """The percentile level of every interval this contrast reports, as a percentage."""

        return int(round(100.0 * (1.0 - self.alpha)))

    def compute_biological_contrasts(
    dataset: Dataset,
    baseline: np.ndarray,
    candidate_matrix: np.ndarray,
    sampling: SamplingManifest,
    design: DesignAudit,
    contrasts: list[BiologicalContrastConfig],
    *,
    run_id: str,
    candidate_id: str,
    donor_factor: str | None = None,
) -> list[MetricRecord]:
    """Assess every prespecified contrast on the frozen partition, before and after.

    ``baseline`` and ``candidate_matrix`` must be aligned with the dataset's cell
    order, as every common-partition metric requires.  One record is produced per
    contrast per frozen population.
    """
    if not contrasts:
        return []

    obs = dataset.obs
    keep = obs[CELL_ID].isin(sampling.labels_by_cell.keys())
    obs = obs.loc[keep].copy()
    obs["_label"] = obs[CELL_ID].map(sampling.labels_by_cell)
    batch_of = dataset.sample_batches()
    obs["_batch"] = obs[SAMPLE_ID].map(batch_of).astype(str)
    positions = {cid: i for i, cid in enumerate(dataset.obs[CELL_ID])}
    obs["_row"] = obs[CELL_ID].map(positions)
    if donor_factor:
        donor_of = {sid: rec.donor_id for sid, rec in dataset.samples.items()}
        obs["_unit"] = obs[SAMPLE_ID].map(donor_of).astype(str)
    else:
        obs["_unit"] = obs[SAMPLE_ID].astype(str)

    populations = sorted(set(sampling.labels_by_cell.values()))
    records: list[MetricRecord] = []

    for contrast in contrasts:
        common = dict(
            run_id=run_id, candidate_id=candidate_id, marker=contrast.marker,
            comparison_id=f"contrast:{contrast.contrast_id}", aggregation_weight=1.0,
        )
        if contrast.marker not in sampling.evaluation_markers:
            records.append(_record(sampling, status=MetricStatus.INELIGIBLE, value=None,
                reason=(
                    f"marker {contrast.marker!r} is not flagged use_for_evaluation; a "
                    "contrast is never computed on an undeclared marker"
                ),
                **common,
            ))
            continue
        col = int(dataset.feature_index([contrast.marker])[0])
        group_of = {sid: _group_level(rec, contrast.group_factor)
                    for sid, rec in dataset.samples.items()}
        obs["_group"] = obs[SAMPLE_ID].map(group_of)

        for pop in populations:
            rows = obs.loc[obs["_label"] == pop]
            rows_a = rows.loc[rows["_group"] == contrast.group_a]
            rows_b = rows.loc[rows["_group"] == contrast.group_b]
            local = dict(common, population=pop)
            n_a, n_b = len(rows_a), len(rows_b)

            if contrast.group_factor in design.confounded_factors:
                records.append(_record(sampling, status=MetricStatus.INELIGIBLE,
                    value=None,
                    reason=(
                        f"unassessable: the contrast factor {contrast.group_factor!r} "
                        f"is perfectly confounded with the correction factor "
                        f"{design.correction_factor!r}; the group contrast and the "
                        "batch contrast are the same contrast, so preservation cannot "
                        "be attributed to the correction and is never reported as "
                        "preserved"
                    ),
                    **local,
                ))
                continue
            batches_a = set(rows_a["_batch"])
            batches_b = set(rows_b["_batch"])
            if not batches_a & batches_b:
                records.append(_record(sampling, status=MetricStatus.INELIGIBLE,
                    value=None,
                    reason=(
                        f"unassessable: groups {contrast.group_a!r} and "
                        f"{contrast.group_b!r} are measured in disjoint batch sets "
                        f"{sorted(batches_a)} and {sorted(batches_b)} for population "
                        f"{pop!r}; the comparison is non-identifiable and is never "
                        "reported as preserved"
                    ),
                    **local,
                ))
                continue
            if n_a < MIN_GROUP_CELLS or n_b < MIN_GROUP_CELLS:
                records.append(_record(sampling, status=MetricStatus.INSUFFICIENT_DATA,
                    value=None,
                    reason=(
                        f"population {pop!r} has {n_a} and {n_b} cells in groups "
                        f"{contrast.group_a!r} and {contrast.group_b!r}; "
                        f"{MIN_GROUP_CELLS} cells per group are required"
                    ),
                    **local,
                ))
                continue

            units_a = sorted(set(rows_a["_unit"]))
            units_b = sorted(set(rows_b["_unit"]))
            if len(units_a) < MIN_UNITS_PER_GROUP or len(units_b) < MIN_UNITS_PER_GROUP:
                kind = "donor(s)" if donor_factor else "sample(s)"
                records.append(_record(sampling, status=MetricStatus.INSUFFICIENT_DATA,
                    value=None,
                    reason=(
                        f"population {pop!r} has {len(units_a)} and {len(units_b)} "
                        f"independent {kind} in the two groups; cells are not "
                        f"independent replicates and {MIN_UNITS_PER_GROUP} per group "
                        "are required"
                    ),
                    **local,
                ))
                continue

            def unit_means(matrix: np.ndarray, rows_grp, units: list[str], col=col) -> np.ndarray:
                v = matrix[rows_grp["_row"].to_numpy(), col]
                u = rows_grp["_unit"].to_numpy()
                return np.array([float(v[u == unit].mean()) for unit in units])

            m_a0 = unit_means(baseline, rows_a, units_a)
            m_b0 = unit_means(baseline, rows_b, units_b)
            m_a1 = unit_means(candidate_matrix, rows_a, units_a)
            m_b1 = unit_means(candidate_matrix, rows_b, units_b)

            shared = sorted(set(units_a) & set(units_b))
            if donor_factor and len(shared) >= 2:
                model = (
                    f"donor-paired difference over {len(shared)} donors measured in "
                    "both groups"
                )
                e0 = float(np.mean([
                    m_b0[units_b.index(d)] - m_a0[units_a.index(d)] for d in shared
                ]))
                e1 = float(np.mean([
                    m_b1[units_b.index(d)] - m_a1[units_a.index(d)] for d in shared
                ]))
            else:
                kind = "donor-level" if donor_factor else "sample-level"
                detail = (
                    "no donor spans both groups" if donor_factor
                    else "no donor factor is declared"
                )
                model = f"pooled {kind} means ({detail})"
                e0 = float(m_b0.mean() - m_a0.mean())
                e1 = float(m_b1.mean() - m_a1.mean())

            change = e1 - e0
            tol = contrast.preservation_tolerance
            preserved = abs(change) <= tol
            records.append(_record(sampling, status=MetricStatus.OK, value=change,
                n_cells=n_a + n_b,
                n_donors=len(set(units_a) | set(units_b)) if donor_factor else None,
                n_samples=len(set(rows_a[SAMPLE_ID]) | set(rows_b[SAMPLE_ID])),
                reason=(
                    f"effect before {e0:.4g}, after {e1:.4g} ({model}); change "
                    f"{change:.4g} is {'within' if preserved else 'beyond'} the "
                    f"predeclared preservation tolerance {tol:.4g} (origin: "
                    f"{contrast.preservation_tolerance_origin}) -> "
                    + ("preserved" if preserved else "not_preserved")
                ),
                **local,
            ))

    return records

    def compute_common_metrics(
    dataset: Dataset,
    candidate_matrix: np.ndarray,
    sampling: SamplingManifest,
    design: DesignAudit,
    *,
    run_id: str,
    candidate_id: str,
) -> MetricResult:
    """All common-namespace metrics for one candidate, on the frozen partition."""

    obs = _grouping(dataset, sampling)
    markers = sampling.evaluation_markers
    if not markers:
        return MetricResult(
            run_id=run_id, candidate_id=candidate_id, metric_group="common",
            status="ineligible",
            reason="no marker is flagged use_for_evaluation in the marker manifest",
        )

    feature_idx = dataset.feature_index(markers)
    baseline = dataset.expression()
    records: list[MetricRecord] = []
    # The residual 'ungated' pool is a heterogeneous leftover, not a biological
    # population: it is reported descriptively (counts, composition) but is excluded
    # from the quantitative population metrics, where it could otherwise dominate the
    # worst-marker summaries that drive the no-effect gate.
    populations = sorted(
        pop for pop in set(sampling.labels_by_cell.values()) if pop != UNGATED
    )
    pairs = _comparisons(design)

    donors_of = {sid: rec.donor_id for sid, rec in dataset.samples.items()}

    # -- distribution alignment within frozen populations ----------------------------
    for pop in populations:
        pop_rows = obs.loc[obs["_label"] == pop]
        for (batch_a, batch_b), comparison_id in sorted(pairs.items()):
            rows_a = pop_rows.loc[pop_rows["_batch"] == batch_a]
            rows_b = pop_rows.loc[pop_rows["_batch"] == batch_b]
            n_a, n_b = len(rows_a), len(rows_b)
            samples = sorted(set(rows_a[SAMPLE_ID]) | set(rows_b[SAMPLE_ID]))
            n_samples_a = rows_a[SAMPLE_ID].nunique()
            n_samples_b = rows_b[SAMPLE_ID].nunique()
            donors = sorted({donors_of.get(s) for s in samples if donors_of.get(s)})
            coverage = float(min(n_a, n_b)) / MIN_GROUP_CELLS if MIN_GROUP_CELLS else 1.0

            for marker in markers:
                common = dict(
                    run_id=run_id, candidate_id=candidate_id, marker=marker, population=pop,
                    comparison_id=comparison_id, n_samples=len(samples),
                    n_donors=len(donors) or None, n_cells=n_a + n_b,
                    coverage=min(coverage, 1.0), aggregation_weight=1.0,
                )
                if n_a < MIN_GROUP_CELLS or n_b < MIN_GROUP_CELLS:
                    records.append(_record(
                        EMD_WITHIN_POPULATION, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=(
                            f"population {pop!r} has {n_a} and {n_b} evaluation cells in "
                            f"{batch_a} and {batch_b}; {MIN_GROUP_CELLS} are required in each"
                        ),
                        **common,
                    ))
                    continue
                if n_samples_a < MIN_SAMPLES_PER_GROUP or n_samples_b < MIN_SAMPLES_PER_GROUP:
                    records.append(_record(
                        EMD_WITHIN_POPULATION, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=(
                            f"population {pop!r} has {n_samples_a} and {n_samples_b} "
                            f"independent sample(s) in {batch_a} and {batch_b}; cells are "
                            f"not independent replicates and {MIN_SAMPLES_PER_GROUP} "
                            "samples per batch are required"
                        ),
                        **common,
                    ))
                    continue
                col = markers.index(marker)
                x = candidate_matrix[rows_a["_row"].to_numpy(), feature_idx[col]]
                y = candidate_matrix[rows_b["_row"].to_numpy(), feature_idx[col]]
                records.append(_record(
                    EMD_WITHIN_POPULATION, sampling, value=float(
                        stats.wasserstein_distance(x, y)
                    ), status=MetricStatus.OK, **common,
                ))

    # -- uncertainty on the EMD records (Stage 8) -------------------------------------
    # Donor-block bootstrap intervals, or an explicit statement of why none could
    # be computed.  Uncertainty is never silently omitted.
    attach_emd_uncertainty(records, dataset, sampling, candidate_matrix,
                           seed=sampling.seed)

    # -- dispersion -------------------------------------------------------------------
    n_compressed = n_total = 0
    for pop in populations:
        pop_rows = obs.loc[obs["_label"] == pop]
        for batch in sorted(set(obs["_batch"])):
            rows = pop_rows.loc[pop_rows["_batch"] == batch]
            idx = rows["_row"].to_numpy()
            for col, marker in enumerate(markers):
                group_samples = sorted(set(rows[SAMPLE_ID]))
                common = dict(
                    run_id=run_id, candidate_id=candidate_id, marker=marker, population=pop,
                    comparison_id=f"batch:{batch}", n_cells=len(idx),
                    n_samples=len(group_samples),
                    n_donors=_donors(group_samples, donors_of),
                    aggregation_weight=1.0,
                )
                if len(idx) < MIN_GROUP_CELLS:
                    records.append(_record(
                        MAD_SIGNED_CHANGE, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=f"{len(idx)} cells, {MIN_GROUP_CELLS} required", **common,
                    ))
                    continue
                if rows[SAMPLE_ID].nunique() < MIN_SAMPLES_PER_GROUP:
                    records.append(_record(
                        MAD_SIGNED_CHANGE, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=(
                            f"{rows[SAMPLE_ID].nunique()} independent sample(s); cells are "
                            f"not independent replicates and {MIN_SAMPLES_PER_GROUP} are "
                            "required"
                        ),
                        **common,
                    ))
                    continue
                f = feature_idx[col]
                mad_base = float(stats.median_abs_deviation(baseline[idx, f], scale="normal"))
                mad_cand = float(
                    stats.median_abs_deviation(candidate_matrix[idx, f], scale="normal")
                )
                delta = mad_cand - mad_base
                n_total += 1
                if delta < -MAD_TOLERANCE:
                    n_compressed += 1
                records.append(_record(
                    MAD_SIGNED_CHANGE, sampling, value=delta, status=MetricStatus.OK,
                    reason=f"MAD baseline {mad_base:.4g}, candidate {mad_cand:.4g}",
                    **common,
                ))

    # -- composition ------------------------------------------------------------------
    freq = frequency_table(dataset, sampling.labels_by_cell)
    batch_of = dataset.sample_batches()
    for pop in populations:
        for (batch_a, batch_b), comparison_id in sorted(pairs.items()):
            samples_a = sorted(s for s in freq if batch_of.get(s) == batch_a)
            samples_b = sorted(s for s in freq if batch_of.get(s) == batch_b)
            fa = [freq[s][pop] for s in samples_a]
            fb = [freq[s][pop] for s in samples_b]
            common = dict(
                run_id=run_id, candidate_id=candidate_id, population=pop,
                comparison_id=comparison_id, n_samples=len(fa) + len(fb),
                n_donors=_donors(samples_a + samples_b, donors_of),
                aggregation_weight=1.0,
            )
            if not fa or not fb:
                records.append(_record(
                    POPULATION_FREQUENCY_SHIFT, sampling,
                    status=MetricStatus.INSUFFICIENT_DATA,
                    reason=f"no samples in one of {batch_a}, {batch_b}", **common,
                ))
                continue
            if len(fa) < MIN_SAMPLES_PER_GROUP or len(fb) < MIN_SAMPLES_PER_GROUP:
                records.append(_record(
                    POPULATION_FREQUENCY_SHIFT, sampling,
                    status=MetricStatus.INSUFFICIENT_DATA,
                    reason=(
                        f"{len(fa)} and {len(fb)} independent sample(s) in {batch_a} and "
                        f"{batch_b}; {MIN_SAMPLES_PER_GROUP} per batch are required, "
                        "because the sample is the independent unit"
                    ),
                    **common,
                ))
                continue
            records.append(_record(
                POPULATION_FREQUENCY_SHIFT, sampling,
                value=float(abs(np.mean(fa) - np.mean(fb))), status=MetricStatus.OK,
                reason="samples are weighted equally, so a high-yield sample cannot dominate",
                **common,
            ))

    # -- joint structure --------------------------------------------------------------
    if len(markers) >= 2:
        for pop in populations:
            pop_rows = obs.loc[obs["_label"] == pop]
            for batch in sorted(set(obs["_batch"])):
                rows = pop_rows.loc[pop_rows["_batch"] == batch]
                idx = rows["_row"].to_numpy()
                group_samples = sorted(set(rows[SAMPLE_ID]))
                common = dict(
                    run_id=run_id, candidate_id=candidate_id, population=pop,
                    comparison_id=f"batch:{batch}", n_cells=len(idx),
                    n_samples=len(group_samples),
                    n_donors=_donors(group_samples, donors_of),
                    aggregation_weight=1.0,
                )
                if len(idx) < MIN_GROUP_CELLS:
                    records.append(_record(
                        JOINT_STRUCTURE, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=f"{len(idx)} cells, {MIN_GROUP_CELLS} required", **common,
                    ))
                    continue
                if rows[SAMPLE_ID].nunique() < MIN_SAMPLES_PER_GROUP:
                    records.append(_record(
                        JOINT_STRUCTURE, sampling, status=MetricStatus.INSUFFICIENT_DATA,
                        reason=(
                            f"{rows[SAMPLE_ID].nunique()} independent sample(s); cells are "
                            f"not independent replicates and {MIN_SAMPLES_PER_GROUP} are "
                            "required"
                        ),
                        **common,
                    ))
                    continue
                c_base = _correlation(baseline[np.ix_(idx, feature_idx)])
                c_cand = _correlation(candidate_matrix[np.ix_(idx, feature_idx)])
                if c_base is None or c_cand is None:
                    records.append(_record(
                        JOINT_STRUCTURE, sampling, status=MetricStatus.UNDEFINED,
                        reason="at least one evaluation marker is constant in this group",
                        **common,
                    ))
                    continue
                tri = np.triu_indices(len(markers), k=1)
                records.append(_record(
                    JOINT_STRUCTURE, sampling,
                    value=float(np.mean(np.abs(c_cand[tri] - c_base[tri]))),
                    status=MetricStatus.OK,
                    reason="mean absolute change in within-population marker correlations",
                    **common,
                ))
    else:
        records.append(_record(
            JOINT_STRUCTURE, sampling, run_id=run_id, candidate_id=candidate_id,
            status=MetricStatus.INELIGIBLE,
            reason="joint structure needs at least two evaluation markers",
        ))

    warnings: list[str] = []
    from batchdx.cyto.metrics.registry import MAD_COMPRESSION

    if n_total:
        records.append(_record(
            MAD_COMPRESSION, sampling, run_id=run_id, candidate_id=candidate_id,
            value=float(n_compressed) / n_total, status=MetricStatus.OK,
            n_cells=None, aggregation_weight=1.0,
            reason=f"{n_compressed} of {n_total} (population, batch, marker) groups lost spread",
        ))
    else:
        # A missing metric is never silently absent: with no qualifying group the
        # compression guardrail is unavailable, and that is reported as such.
        records.append(_record(
            MAD_COMPRESSION, sampling, run_id=run_id, candidate_id=candidate_id,
            value=None, status=MetricStatus.INSUFFICIENT_DATA,
            n_cells=None, aggregation_weight=1.0,
            reason="no (population, batch, marker) group met the cell and sample minima, "
                   "so a compression fraction cannot be computed",
        ))

    n_failed = sum(1 for r in records if r.status is MetricStatus.FAILED)
    status = "ok" if not n_failed else "partial"
    return MetricResult(
        run_id=run_id, candidate_id=candidate_id, metric_group="common",
        records=records, status=status, warnings=warnings,
    )

    def compute_reductions(
    results: dict[str, MetricResult],
    sampling: SamplingManifest,
    *,
    run_id: str,
    identity_candidate_id: str,
) -> list[MetricRecord]:
    """Relative EMD reduction per candidate against the identity candidate."""

    if identity_candidate_id not in results:
        return [MetricRecord(
            run_id=run_id, candidate_id="*", metric_namespace="common",
            metric_name=EMD_REDUCTION.name, metric_version=EMD_REDUCTION.version,
            stage=EMD_REDUCTION.stage, status=MetricStatus.INELIGIBLE,
            direction=EMD_REDUCTION.direction, scale=EMD_REDUCTION.scale,
            reason=(
                f"the identity candidate {identity_candidate_id!r} produced no metrics, so "
                "no relative reduction is defined for any candidate"
            ),
        )]

    by_candidate: dict[str, dict[tuple, MetricRecord]] = defaultdict(dict)
    for candidate_id, result in results.items():
        for record in result.records:
            if record.metric_name != EMD_WITHIN_POPULATION.name:
                continue
            by_candidate[candidate_id][_key(record)] = record

    identity = by_candidate.get(identity_candidate_id, {})
    out: list[MetricRecord] = []

    for candidate_id in sorted(by_candidate):
        candidate = by_candidate[candidate_id]
        usable = [
            k for k in identity
            if k in candidate
            and identity[k].status is MetricStatus.OK
            and candidate[k].status is MetricStatus.OK
            and identity[k].value is not None
            and candidate[k].value is not None
        ]
        # Coverage is the candidate's own success rate on the comparisons the identity
        # candidate made eligible.  The identity's own failures are reported separately
        # so a deflated coverage cannot be misread as this candidate's failure.
        n_eligible = sum(
            1 for k in identity if identity[k].status is MetricStatus.OK
        )
        identity_ok = n_eligible
        coverage = (len(usable) / n_eligible) if n_eligible else 0.0
        base = dict(
            run_id=run_id, candidate_id=candidate_id, metric_namespace="common",
            metric_name=EMD_REDUCTION.name, metric_version=EMD_REDUCTION.version,
            stage=EMD_REDUCTION.stage, direction=EMD_REDUCTION.direction,
            scale=EMD_REDUCTION.scale, coverage=coverage,
            label_tier=sampling.label_source, label_confidence=sampling.label_confidence,
            representation=sampling.representation,
        )
        if not usable:
            out.append(MetricRecord(
                **base, status=MetricStatus.INSUFFICIENT_DATA,
                reason="no comparison succeeded for both this candidate and the identity "
                       "candidate",
            ))
            continue
        denominator = sum(float(identity[k].value) for k in usable)
        numerator = denominator - sum(float(candidate[k].value) for k in usable)
        if denominator <= DENOMINATOR_FLOOR:
            out.append(MetricRecord(
                **base, status=MetricStatus.UNDEFINED,
                reason=(
                    f"the identity candidate's total EMD over the shared comparisons is "
                    f"{denominator:.3g}; a relative reduction is undefined and is not "
                    "stabilised with an epsilon"
                ),
            ))
            continue
        out.append(MetricRecord(
            **base, value=float(numerator / denominator), status=MetricStatus.OK,
            reason=(
                f"over {len(usable)} of {n_eligible} comparisons eligible for the identity "
                f"candidate (identity itself succeeded on {identity_ok}); coverage is this "
                "candidate's success rate on the shared set"
            ),
        ))
    return out

    def resolve_contrast(
    dataset: Dataset,
    design: DesignAudit,
    sampling: SamplingManifest,
    spec: ContrastSpec,
) -> ContrastPlan | ContrastRefusal:
    """Decide whether a prespecified contrast may be run, and with which replicate unit.

    Every branch that refuses names the condition that failed.  A contrast is never
    dropped without a record, and never reported as a score of zero.
    """

    values = _factor_values(dataset, spec.factor)
    levels_present = sorted({v for v in values.values() if v is not None})
    if not levels_present:
        return ContrastRefusal(
            spec, "unknown_factor",
            f"no sample declares {spec.factor!r}, so the contrast has no data to run on; "
            f"the declared levels were {spec.level_a!r} and {spec.level_b!r}",
        )

    missing = [lvl for lvl in (spec.level_a, spec.level_b) if lvl not in levels_present]
    if missing:
        return ContrastRefusal(
            spec, "levels_absent",
            f"level(s) {missing} of contrast {spec.contrast_id!r} are not present in "
            f"{spec.factor!r}; the sample table carries {levels_present}. A contrast is "
            "prespecified, not discovered after the fact.",
        )

    # -- identifiability ---------------------------------------------------------------
    if spec.factor not in design.protected_factors:
        return ContrastRefusal(
            spec, "factor_not_protected",
            f"{spec.factor!r} is not among the design's declared protected factors "
            f"{list(design.protected_factors)}, so a difference in it is not separately "
            "estimable from the batch effect; declare it in design.protect first",
        )
    if spec.factor in design.confounded_factors:
        return ContrastRefusal(
            spec, "non_identifiable",
            f"the design audit reports {spec.factor!r} perfectly confounded with "
            f"{design.correction_factor!r} (confounded_factors="
            f"{list(design.confounded_factors)}); the batch term and the {spec.factor!r} "
            "term span the same subspace, so no covariate adjustment separates them and no "
            "preservation claim can rest on this contrast",
        )
    batch_values = _factor_values(dataset, design.correction_factor)
    level_by_batch: dict[str, set[str]] = {}
    for sid, level in values.items():
        if level in {spec.level_a, spec.level_b} and batch_values.get(sid):
            level_by_batch.setdefault(str(batch_values[sid]), set()).add(str(level))
    if len(level_by_batch) > 1 and all(len(v) == 1 for v in level_by_batch.values()):
        return ContrastRefusal(
            spec, "non_identifiable",
            f"the two levels of {spec.factor!r} occupy disjoint batches "
            f"({ {b: sorted(v)[0] for b, v in sorted(level_by_batch.items())} }); the "
            "biological contrast and the batch contrast are the same contrast",
        )

    # -- declared covariates -----------------------------------------------------------
    for covariate in spec.protected_covariates:
        covariate_values = _factor_values(dataset, covariate)
        present = {v for v in covariate_values.values() if v is not None}
        if not present:
            return ContrastRefusal(
                spec, "covariate_absent",
                f"protected covariate {covariate!r} is declared on this contrast but no "
                "sample carries it, so the contrast would not be conditional on anything",
            )
        if present <= {spec.level_a, spec.level_b}:
            return ContrastRefusal(
                spec, "covariate_nested_in_contrast",
                f"protected covariate {covariate!r} is nested inside {spec.factor!r}: its "
                f"levels {sorted(present)} are a subset of "
                f"{{{spec.level_a!r}, {spec.level_b!r}}}, so the contrast and the "
                "covariate cannot be separated",
            )

    # -- scope -------------------------------------------------------------------------
    evaluation_markers = tuple(sampling.evaluation_markers)
    markers = evaluation_markers if spec.markers is None else tuple(spec.markers)
    if not markers:
        return ContrastRefusal(
            spec, "no_evaluation_markers",
            "the frozen partition names no marker flagged use_for_evaluation, so there is "
            "nothing to test",
        )
    absent_markers = [m for m in markers if m not in evaluation_markers]
    if absent_markers:
        return ContrastRefusal(
            spec, "marker_absent",
            f"marker(s) {absent_markers} are not flagged use_for_evaluation on the frozen "
            f"partition, which carries {list(evaluation_markers)}",
        )
    frozen_populations = sorted(set(sampling.labels_by_cell.values()))
    populations = (
        tuple(frozen_populations) if spec.populations is None else tuple(spec.populations)
    )
    absent_populations = [p for p in populations if p not in frozen_populations]
    if not populations or absent_populations:
        return ContrastRefusal(
            spec, "population_absent",
            f"population(s) {absent_populations or list(populations)} are not on the frozen "
            f"partition, which carries {frozen_populations}",
        )

    # -- replicate unit ----------------------------------------------------------------
    contributing = tuple(sorted(
        sid for sid, level in values.items() if level in {spec.level_a, spec.level_b}
    ))
    donors = {sid: rec.donor_id for sid, rec in dataset.samples.items()}
    has_donor = all(donors.get(sid) for sid in contributing)
    # The replicate unit is the donor when every contributing sample declares one, and
    # the sample itself otherwise.  Mixing the two would make a resampled block
    # sometimes a donor and sometimes a sample.
    block_of = {sid: (str(donors[sid]) if has_donor else sid) for sid in contributing}
    block_unit = "donor_id" if has_donor else "sample_id"
    level_of = {sid: str(values[sid]) for sid in contributing}

    levels_in_block: dict[str, set[str]] = {}
    for sid in contributing:
        levels_in_block.setdefault(block_of[sid], set()).add(level_of[sid])
    complete = sorted(
        block for block, levels in levels_in_block.items()
        if levels == {spec.level_a, spec.level_b}
    )
    paired = has_donor and len(complete) >= MIN_PAIRED_DONORS
    if spec.model == "paired" and not paired:
        return ContrastRefusal(
            spec, "pairing_unsupported",
            f"the contrast declares a paired model but only {len(complete)} donor(s) "
            f"contribute both {spec.level_a!r} and {spec.level_b!r}; at least "
            f"{MIN_PAIRED_DONORS} complete donor pairs are required, and a donor factor is "
            + ("declared for every contributing sample" if has_donor
               else "not declared for every contributing sample"),
        )
    model = "paired" if paired else "unpaired"
    rows = tuple(complete) if paired else tuple(sorted(levels_in_block))
    index = {row: i for i, row in enumerate(rows)}

    if paired:
        # Each block is one donor holding both levels, so relabelling inside it is a swap
        # of that donor's two values, and every donor has exactly two such assignments.
        blocks = tuple((f"donor:{row}", (index[row],)) for row in rows)
        reachable = 2 ** len(rows)
    else:
        # Each block is a batch, and relabelling inside it permutes the labels among that
        # batch's own samples, so the number of each level in the batch cannot change.
        grouped: dict[str, list[int]] = {}
        for sid in contributing:
            grouped.setdefault(str(batch_values.get(sid)), []).append(index[block_of[sid]])
        blocks = tuple((f"batch:{b}", tuple(sorted(v))) for b, v in sorted(grouped.items()))
        holds_a = {
            row: any(level_of[s] == spec.level_a for s in contributing if block_of[s] == row)
            for row in rows
        }
        reachable = 1
        for _, indices in blocks:
            reachable *= math.comb(len(indices), sum(1 for i in indices if holds_a[rows[i]]))
            reachable = min(reachable, 10**9)

    return ContrastPlan(
        spec=spec,
        level_a=spec.level_a,
        level_b=spec.level_b,
        model=model,
        block_unit=block_unit,
        rows=rows,
        blocks=blocks,
        reachable_permutations=min(reachable, 10**9),
        contributing=contributing,
        block_of=block_of,
        level_of=level_of,
        markers=markers,
        populations=populations,
    )

    def attach_emd_uncertainty(
    records: list[MetricRecord],
    dataset: Dataset,
    sampling: SamplingManifest,
    candidate_matrix: np.ndarray,
    *,
    seed: int,
    n_resamples: int = BOOTSTRAP_RESAMPLES,
) -> None:
    """Attach donor-block bootstrap CIs to ``emd_within_population`` records, in place.

    A record whose comparison has at least two donors per batch gains
    ``uncertainty_low``/``uncertainty_high`` and ``uncertainty_kind =
    'donor_block_bootstrap_ci'``.  A record whose comparison does not keeps its value
    but states in the reason that no interval could be computed and why.  Uncertainty
    is never silently omitted and never replaced by a zero-width interval.
    """
    obs = dataset.obs
    keep = obs[CELL_ID].isin(sampling.labels_by_cell.keys())
    obs = obs.loc[keep].copy()
    obs["_label"] = obs[CELL_ID].map(sampling.labels_by_cell)
    batch_of = dataset.sample_batches()
    obs["_batch"] = obs[SAMPLE_ID].map(batch_of).astype(str)
    positions = {cid: i for i, cid in enumerate(dataset.obs[CELL_ID])}
    obs["_row"] = obs[CELL_ID].map(positions)
    donor_of = {sid: rec.donor_id for sid, rec in dataset.samples.items()}
    obs["_donor"] = obs[SAMPLE_ID].map(donor_of).astype(str)

    markers = sampling.evaluation_markers
    if not markers:
        return
    feature_idx = dataset.feature_index(markers)
    col_of = {m: i for i, m in enumerate(markers)}

    for record in records:
        if record.metric_name != "emd_within_population":
            continue
        if record.status is not MetricStatus.OK:
            continue
        if record.population is None or record.marker is None:
            continue
        if record.comparison_id is None or "__vs__" not in record.comparison_id:
            continue
        batch_a, batch_b = record.comparison_id.split("__vs__", 1)
        rows = obs.loc[obs["_label"] == record.population]
        ra = rows.loc[rows["_batch"] == batch_a]
        rb = rows.loc[rows["_batch"] == batch_b]
        n_donors_a = ra["_donor"].nunique()
        n_donors_b = rb["_donor"].nunique()
        if n_donors_a < MIN_DONORS_PER_GROUP or n_donors_b < MIN_DONORS_PER_GROUP:
            record.reason = (
                (record.reason + "; " if record.reason else "")
                + f"no donor-block bootstrap CI: this comparison has {n_donors_a} "
                f"donor(s) in {batch_a} and {n_donors_b} in {batch_b}, and at least "
                f"{MIN_DONORS_PER_GROUP} donors per batch are required to resample"
            )
            continue
        col = feature_idx[col_of[record.marker]]
        ci = donor_block_bootstrap_emd(
            candidate_matrix[ra["_row"].to_numpy(), col],
            candidate_matrix[rb["_row"].to_numpy(), col],
            ra["_donor"].to_numpy(),
            rb["_donor"].to_numpy(),
            n_resamples=n_resamples, seed=seed,
        )
        if ci is None:  # pragma: no cover - guarded by the donor count above
            continue
        record.uncertainty_low = ci[0]
        record.uncertainty_high = ci[1]
        record.uncertainty_kind = BOOTSTRAP_KIND
        record.reason = (
            (record.reason + "; " if record.reason else "")
            + f"donor-block bootstrap CI [{ci[0]:.4g}, {ci[1]:.4g}] "
            f"({BOOTSTRAP_ORIGIN})"
        )

    def donor_block_bootstrap_emd(
    x: np.ndarray,
    y: np.ndarray,
    donors_x: np.ndarray,
    donors_y: np.ndarray,
    *,
    n_resamples: int = BOOTSTRAP_RESAMPLES,
    seed: int,
    level: float = 0.95,
) -> tuple[float, float] | None:
    """Percentile CI for the two-sample EMD, resampling donors within each group.

    ``x`` and ``y`` are the two batches' cell values on one marker; ``donors_x`` and
    ``donors_y`` give each cell's donor.  Donors are resampled with replacement within
    their own batch, never across batches.  Returns ``None`` when either group has
    fewer than :data:`MIN_DONORS_PER_GROUP` donors.
    """
    ux = np.unique(donors_x)
    uy = np.unique(donors_y)
    if len(ux) < MIN_DONORS_PER_GROUP or len(uy) < MIN_DONORS_PER_GROUP:
        return None
    rng = np.random.default_rng(seed)
    idx_x = {d: np.flatnonzero(donors_x == d) for d in ux}
    idx_y = {d: np.flatnonzero(donors_y == d) for d in uy}
    drawn = np.empty(n_resamples, dtype=float)
    for b in range(n_resamples):
        pick_x = rng.choice(len(ux), size=len(ux), replace=True)
        pick_y = rng.choice(len(uy), size=len(uy), replace=True)
        cells_x = np.concatenate([idx_x[ux[i]] for i in pick_x])
        cells_y = np.concatenate([idx_y[uy[i]] for i in pick_y])
        drawn[b] = stats.wasserstein_distance(x[cells_x], y[cells_y])
    alpha = (1.0 - level) / 2.0
    lo, hi = np.quantile(drawn, [alpha, 1.0 - alpha])
    return float(lo), float(hi)

    def restricted_permutation_test(
    values: np.ndarray,
    batch_labels: np.ndarray,
    strata: np.ndarray,
    *,
    n_permutations: int = PERMUTATION_RESAMPLES,
    seed: int,
) -> PermutationResult:
    """Two-sided permutation test for a batch difference in means, within strata.

    Batch labels are permuted only inside a stratum (a donor block at fixed
    protected-factor levels).  A stratum contributes exchanges only when it contains
    both batch labels; every other stratum is held fixed.  The test therefore never
    swaps labels across donor blocks and never across different protected levels —
    when no stratum holds both labels there is no valid exchange, and the p-value is
    reported as unavailable rather than as 1.0.
    """
    values = np.asarray(values, dtype=float)
    labels = np.asarray(batch_labels, dtype=object)
    strata = np.asarray(strata, dtype=object)
    levels = sorted(set(labels.tolist()))
    if len(levels) != 2:
        return PermutationResult(
            observed=float("nan"), p_value=None, n_permutations=0,
            n_exchangeable_strata=0,
            reason=f"permutation test needs exactly two batch levels, got {levels}",
        )
    a, b = levels
    mask_a = labels == a
    if not mask_a.any() or not (~mask_a).any():
        return PermutationResult(
            observed=float("nan"), p_value=None, n_permutations=0,
            n_exchangeable_strata=0,
            reason="one batch level has no values; nothing is exchangeable",
        )
    observed = float(values[mask_a].mean() - values[~mask_a].mean())

    idx_by_key = {k: np.flatnonzero(strata == k) for k in np.unique(strata)}
    exchangeable = [
        idx for idx in idx_by_key.values()
        if set(labels[idx].tolist()) == {a, b}
    ]
    if not exchangeable:
        return PermutationResult(
            observed=observed, p_value=None, n_permutations=0, n_exchangeable_strata=0,
            reason=(
                "no stratum contains both batch labels, so no exchange is possible "
                "within donor blocks at fixed protected-factor levels; a permutation "
                "test would either do nothing or permute across different protected "
                "levels, and neither is valid"
            ),
        )

    rng = np.random.default_rng(seed)
    extreme = 0
    for _ in range(n_permutations):
        perm = labels.copy()
        for idx in exchangeable:
            perm[idx] = rng.permutation(labels[idx])
        stat = values[perm == a].mean() - values[perm != a].mean()
        if abs(stat) >= abs(observed):
            extreme += 1
    p_value = (1.0 + extreme) / (1.0 + n_permutations)
    return PermutationResult(
        observed=observed, p_value=float(p_value), n_permutations=n_permutations,
        n_exchangeable_strata=len(exchangeable),
    )
